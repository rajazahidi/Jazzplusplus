cmake_minimum_required(VERSION 3.16)
project(jazzplusplus VERSION 5.4.16 LANGUAGES C CXX)

    add_compile_options(-Wall -Wno-unknown-pragmas -Wno-deprecated-declarations)
elseif(MSVC)
    add_compile_options(/W3 /wd4996)
endif()

# -----------------------------------------------------------------------------
# Modern JUCE 8 Application (jazz-juce)
# -----------------------------------------------------------------------------
option(BUILD_JAZZ_JUCE "Build modern JUCE-based Jazz++ application" ON)
if(BUILD_JAZZ_JUCE)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/libs/JUCE/CMakeLists.txt")
        add_subdirectory(libs/JUCE)
    else()
        include(FetchContent)
        FetchContent_Declare(JUCE
            GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
            GIT_TAG 8.0.6
            GIT_SHALLOW TRUE
        )
        FetchContent_MakeAvailable(JUCE)
    endif()

    juce_add_gui_app(jazz-juce
        PRODUCT_NAME "Jazz++ JUCE"
        VERSION 6.0.0
    )

    target_sources(jazz-juce PRIVATE
        src/juce/JazzJuceApp.cpp
        src/juce/MainComponent.cpp
        src/juce/MainComponent.h
        src/juce/JazzLookAndFeel.h
        src/juce/AudioMidiEngine.cpp
        src/juce/AudioMidiEngine.h
        src/juce/ClapPluginFormat.cpp
        src/juce/ClapPluginFormat.h
        src/juce/PluginHostComponent.cpp
        src/juce/PluginHostComponent.h
        src/juce/ArrangerComponent.cpp
        src/juce/ArrangerComponent.h
        src/juce/PianoRollComponent.cpp
        src/juce/PianoRollComponent.h
        src/juce/DrumMachineComponent.cpp
        src/juce/DrumMachineComponent.h
        src/juce/FretboardComponent.cpp
        src/juce/FretboardComponent.h
        src/juce/AudioEffectsRackComponent.cpp
        src/juce/AudioEffectsRackComponent.h
        src/AudioEffects.cpp
        src/SoundGenerator.cpp
        src/MidiEffects.cpp
        src/ChordScaleData.cpp
    )

    target_compile_definitions(jazz-juce PRIVATE
        JAZZ_UNIT_TEST=1
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_PLUGINHOST_VST3=1
    )

    target_include_directories(jazz-juce PRIVATE
        src
        src/juce
        libs/clap/include
    )

    target_compile_features(jazz-juce PRIVATE cxx_std_20)

    target_link_libraries(jazz-juce PRIVATE
        juce::juce_audio_basics
        juce::juce_audio_devices
        juce::juce_audio_formats
        juce::juce_audio_processors
        juce::juce_audio_utils
        juce::juce_dsp
        juce::juce_gui_basics
        juce::juce_gui_extra
    )

    add_custom_command(TARGET jazz-juce POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E create_symlink
            "$<TARGET_FILE:jazz-juce>"
            "${CMAKE_BINARY_DIR}/jazz-juce"
        COMMENT "Creating convenience symlink build/jazz-juce"
    )

    install(TARGETS jazz-juce RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
