cmake_minimum_required(VERSION 3.16)
project(jazzplusplus VERSION 5.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Compiler warning flags
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wno-unknown-pragmas -Wno-deprecated-declarations)
elseif(MSVC)
    add_compile_options(/W3 /wd4996)
endif()

if(WIN32)
    enable_language(RC)
    set(wxWidgets_EXCLUDE_COMMON_LIBRARIES TRUE)
    set(BUILD_SHARED_LIBS ON)
    if(DEFINED wxWidgets_ROOT_DIR)
        set(WX_ROOT_DIR ${wxWidgets_ROOT_DIR} CACHE INTERNAL "wxWidgets_ROOT_DIR")
    endif()
endif()

# Find wxWidgets
find_package(wxWidgets REQUIRED core base adv net html)
include(${wxWidgets_USE_FILE})

if(WIN32)
    # Windows configuration
    set(PLATFORM_SOURCES
        src/mswin/WindowsAudioInterface.cpp
        src/mswin/WindowsConsole.cpp
        src/mswin/WindowsMidiInterface.cpp
        src/mswin/WindowsPlayer.cpp
        src/mswin/jazz.rc
    )

    set(PLATFORM_LIBRARIES
        winmm
        ws2_32
    )

    set(PLATFORM_INCLUDES
        ${CMAKE_CURRENT_SOURCE_DIR}/src/mswin
    )

    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/config.h
"// Generated by CMake for Jazz++ (Windows)
#pragma once

#define HAVE_CONFIG_H 1
#define PACKAGE \"jazzplusplus\"
#define VERSION \"5.0.0\"
")

    add_definitions(-DHAVE_CONFIG_H -DNOMINMAX -DWIN32_LEAN_AND_MEAN)
    set(GUI_TYPE WIN32)
else()
    # Linux / ALSA configuration
    find_package(ALSA REQUIRED)

    set(PLATFORM_SOURCES
        src/AlsaDriver.cpp
        src/AlsaPlayer.cpp
        src/AlsaThru.cpp
        src/AudioDriver.cpp
    )

    set(PLATFORM_LIBRARIES
        ${ALSA_LIBRARIES}
        pthread
        m
    )

    set(PLATFORM_INCLUDES
        ${ALSA_INCLUDE_DIRS}
    )

    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/config.h
"// Generated by CMake for Jazz++ (Linux)
#pragma once

#define HAVE_CONFIG_H 1
#define DEV_ALSA 1
#define DEV_SEQUENCER2 1
#define PACKAGE \"jazzplusplus\"
#define VERSION \"5.0.0\"
")

    add_definitions(
        -DHAVE_CONFIG_H
        -DDEV_ALSA
        -DDEV_SEQUENCER2
        -DJAZZ_DATADIR="${CMAKE_INSTALL_PREFIX}/share/jazz"
    )

    set(GUI_TYPE "")
endif()

# Include directories
include_directories(

)

target_link_libraries(jazz PRIVATE
    ${wxWidgets_LIBRARIES}
    ${PLATFORM_LIBRARIES}
)

include(GNUInstallDirs)
install(TARGETS jazz RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS scripts/jazz-launcher.sh RENAME jazz-launcher DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES jazz.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES resources/icons/jazz.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES resources/icons/jazz.png DESTINATION ${CMAKE_INSTALL_DATADIR}/pixmaps)

foreach(size 16 32 48 64 128 256)
    install(FILES resources/icons/jazz-${size}.png
            DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps
            RENAME jazz.png)
endforeach()

install(DIRECTORY conf/ DESTINATION ${CMAKE_INSTALL_DATADIR}/jazz/conf)
install(DIRECTORY song/ DESTINATION ${CMAKE_INSTALL_DATADIR}/jazz/song)
